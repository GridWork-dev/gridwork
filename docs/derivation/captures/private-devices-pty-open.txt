    import ctypes
    import errno
    import os


    def outcome(label, operation):
        try:
            fd = operation()
        except OSError as error:
            name = errno.errorcode.get(error.errno, "UNKNOWN")
            print(f"{label}: errno {error.errno} ({name})")
        else:
            os.close(fd)
            print(f"{label}: success")


    for path in ["/dev/ptmx", "/dev/pts", "/dev/pts/ptmx"]:
        print(f"exists {path}: {'yes' if os.path.exists(path) else 'no'}")

    libc = ctypes.CDLL(None, use_errno=True)
    libc.posix_openpt.argtypes = [ctypes.c_int]
    libc.posix_openpt.restype = ctypes.c_int


    def posix_openpt():
        fd = libc.posix_openpt(os.O_RDWR | os.O_NOCTTY)
        if fd < 0:
            value = ctypes.get_errno()
            raise OSError(value, os.strerror(value))
        return fd


    outcome("posix_openpt", posix_openpt)
    outcome("open /dev/ptmx", lambda: os.open("/dev/ptmx", os.O_RDWR | os.O_NOCTTY))
